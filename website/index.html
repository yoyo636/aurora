Python · 10 步

import json, urllib.request
req = urllib.request.Request(url)
resp = urllib.request.urlopen(req)
data = json.loads(resp.read().decode())
names = [u["name"] for u in data["users"]]
unique = list(dict.fromkeys(names))
ages = [u["age"] for u in data["users"]]
avg = sum(ages) / len(ages)
with open("out.json","w") as f: json.dump(...,f)
print(f"完成: {len(names)}")

Aurora · 4 步

data = json(get(url))
names = unique(pluck(data["users"], "name"))
avg = avg(pluck(data["users"], "age"))
write("out.json", to_json({names, avg}))
